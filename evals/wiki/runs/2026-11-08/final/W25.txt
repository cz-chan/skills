```ts
recipes.delete("/:id", async (c) => {
  const rows = await sql`
    delete from recipes where id = ${c.req.param("id")} returning id`;
  if (rows.length === 0) return c.json({ error: "not found" }, 404);
  return c.body(null, 204);
});
```
